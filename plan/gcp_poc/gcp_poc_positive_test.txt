2) Create the IAM policy manually in dev account:

cat > /tmp/gcp-trust-policy.json << 'EOF'
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Sid": "DataprocMec2prod6ReadCtkbTestBucket",
      "Effect": "Allow",
      "Action": [

EOF
aws-okta exec dev-admin -- aws iam create-policy \
  --policy-name dataproc-mec2prod6-gcp-ctkb-test \
  --policy-document file:///tmp/gcp-trust-policy.json \
  --description "Allows mec2prod6 Dataproc GCP SA via OIDC to read vyom-test-cross-provider"

3) Create trust policy
cat > /tmp/gcp-trust-role.json << 'EOF'
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Effect": "Allow",
      "Principal": {
        "Federated": "arn:aws:iam::125952381355:oidc-provider/accounts.google.com"
      },
      "Action": "sts:AssumeRoleWithWebIdentity",
      "Condition": {
        "StringEquals": {
          "accounts.google.com:aud": "111695966871158907715",
          "accounts.google.com:sub": "111695966871158907715"
        }
      }
    }
  ]
}
EOF

4) Phase 3e — Create the IAM role:

aws-okta exec dev-admin -- aws iam create-role \
  --role-name dataproc_mec2prod6_gcp_trust_role \
  --assume-role-policy-document file:///tmp/gcp-trust-role.json \
  --description "Assumed by mec2prod6 Dataproc GCP SA via OIDC to access vyom-test-cross-provider"

5) attach policy to role
aws-okta exec dev-admin -- aws iam attach-role-policy \
  --role-name dataproc_mec2prod6_gcp_trust_role \
  --policy-arn arn:aws:iam::125952381355:policy/dataproc-mec2prod6-gcp-ctkb-test


6) ssh back to node
gcloud compute ssh --zone "me-central2-a" "af2-action-and-suggestion-w-0" \
